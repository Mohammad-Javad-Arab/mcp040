        MessageBox(hDlg, L"Parent listbox not found!", L"Error", MB_OK | MB_ICONERROR);
        return;

    HWND hListView = GetDlgItem(hDlg, IDC_TERMINAL_LIST4); // Get ListView handle in current dialog

    // Clear existing items in the ListView
    //ListView_DeleteAllItems(hListView);

    // Get the count of items in the parent listbox

        MessageBox(hDlg, L"Failed to get item count from parent listbox!", L"Error", MB_OK | MB_ICONERROR);

    }

    for (int i = 0; i < itemCount; i++) {

        SendMessage(hParentListBox, LB_GETTEXT, i, (LPARAM)itemText); // Get text from the parent listbox

    // Function for receiving data in a separate thread
    static void ReceiveDataThread(SOCKET ConnectSocket, HWND hWndLogBox) {
        char recvbuf[1024];
        int iResult;

        while (true) {
            iResult = recv(ConnectSocket, recvbuf, sizeof(recvbuf), 0);
            if ((iResult > 0) && (iResult < 1024)) {
                recvbuf[iResult] = '\0'; // Null-terminate the received data
                int wideLen = MultiByteToWideChar(CP_ACP, 0, recvbuf, -1, NULL, 0);
                if (wideLen > 0) {
                    wchar_t* wideMessage = new wchar_t[wideLen];
                    MultiByteToWideChar(CP_ACP, 0, recvbuf, -1, wideMessage, wideLen);

                    if (hWndLogBox) {
                        int logLen = GetWindowTextLength(hWndLogBox);

                        // Get the existing content
                        wchar_t* currentText = new wchar_t[logLen + 1];
                        GetWindowTextW(hWndLogBox, currentText, logLen + 1);

                        // Concatenate the new message
                        size_t newSize = logLen + wcslen(wideMessage) + 3; // Include space for "\r\n"
                        wchar_t* newText = new wchar_t[newSize];
                        swprintf_s(newText, newSize, L"%s\r\n%s", currentText, wideMessage);

                        // Update the log box
                        SetWindowTextW(hWndLogBox, newText);

                        // Clean up
                        delete[] currentText;
                        delete[] newText;
                    }
                    else {
                        MessageBox(NULL, L"hWndLogBox is invalid\n", L"Error", MB_OK);
                    }

                    delete[] wideMessage;
                }


                
                
            }
            else if (iResult == 0) {
                break; // Connection closed
            }
            else {
                MessageBox(NULL, L"Receive failed", L"Error", MB_OK);
                break;
            }
        }
    }

    int APIENTRY wWinMain(_In_ HINSTANCE hInstance,
        _In_opt_ HINSTANCE hPrevInstance,
        _In_ LPWSTR    lpCmdLine,
        _In_ int       nCmdShow)
    {
        UNREFERENCED_PARAMETER(hPrevInstance);
        UNREFERENCED_PARAMETER(lpCmdLine);

        // Initialize global strings
        LoadStringW(hInstance, IDS_APP_TITLE, szTitle, MAX_LOADSTRING);
        LoadStringW(hInstance, IDC_MCP040, szWindowClass, MAX_LOADSTRING);
        MyRegisterClass(hInstance);

        INITCOMMONCONTROLSEX icex;
        icex.dwSize = sizeof(INITCOMMONCONTROLSEX);
        icex.dwICC = ICC_LISTVIEW_CLASSES; // Initialize ListView controls
        InitCommonControlsEx(&icex);
       


        // Perform application initialization:
        if (!InitInstance(hInstance, nCmdShow))
        {
            return FALSE;
        }

        HACCEL hAccelTable = LoadAccelerators(hInstance, MAKEINTRESOURCE(IDC_MCP040));

        MSG msg;

        // Initialize Winsock
        WSADATA wsaData;
        int iResult = WSAStartup(MAKEWORD(2, 2), &wsaData);
        if (iResult != 0) {
            MessageBox(NULL, L"WSAStartup failed", L"Error", MB_OK);
            return FALSE;
        }

        // Create a socket
        ConnectSocket = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
        if (ConnectSocket == INVALID_SOCKET) {
            MessageBox(NULL, L"Socket creation failed", L"Error", MB_OK);
            WSACleanup();
            return FALSE;
        }

        // Setup the server address structure
        sockaddr_in serverAddr{};
        serverAddr.sin_family = AF_INET;
        serverAddr.sin_port = htons(serverPort);
        inet_pton(AF_INET, serverAddress, &serverAddr.sin_addr);

        // Connect to server
        iResult = connect(ConnectSocket, (sockaddr*)&serverAddr, sizeof(serverAddr));
        if (iResult == SOCKET_ERROR) {
            MessageBox(NULL, L"Connection failed", L"Error", MB_OK);
            closesocket(ConnectSocket);
            WSACleanup();
            return FALSE;
        }

        // Start the receive data thread
        std::thread receiveThread(ReceiveDataThread, ConnectSocket, hWndLogBox);
        receiveThread.detach(); // Detach the thread so it runs independently

        // Main message loop:
        while (GetMessage(&msg, nullptr, 0, 0))
        {
            if (!TranslateAccelerator(msg.hwnd, hAccelTable, &msg))
            {
                TranslateMessage(&msg);
                DispatchMessage(&msg);
            }
        }

        // Cleanup
        closesocket(ConnectSocket);
        WSACleanup();

        return (int)msg.wParam;
    }

    // Message handler for about box.
    INT_PTR CALLBACK About(HWND hDlg, UINT message, WPARAM wParam, LPARAM lParam)
    {
        UNREFERENCED_PARAMETER(lParam);
        switch (message)
        {
        case WM_INITDIALOG:
            return (INT_PTR)TRUE;

        case WM_COMMAND:
            if (LOWORD(wParam) == IDOK || LOWORD(wParam) == IDCANCELABOUT)
            {
                EndDialog(hDlg, LOWORD(wParam));
                return (INT_PTR)TRUE;
            }
            break;
        }
        return (INT_PTR)FALSE;
    }
